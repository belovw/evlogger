# Формат строки:  имя ; ответчик(или -) ; команда ; формула
# Ответчик вида 7A1>7A9 = личный запрос блоку 7A1, ответ с 7A9 (так читается батарея).
# Запрос уходит всем блокам (адрес 7DF), а ответ принимается только от блока с указанным адресом
# (команда ATCRA). Адрес ответчика: 7E8 = блок ДВС, 7EA = блок с SOC. "-" значит принимать от всех.
# Формула: переменные A,B,C,D (байты ответа), + - * / и скобки.
# Строки с # игнорируются.
#
# --- Блок ДВС (отвечает с 7E8), стандартные PID Mode 01 ---
speed_kmh ; 7E8 ; 010D ; A
coolant_c ; 7E8 ; 0105 ; A-40
rpm ; 7E8 ; 010C ; (A*256+B)/4
fuel_pct ; 7E8 ; 012F ; A*100/255
ambient_c ; 7E8 ; 0146 ; A-40
load_pct ; 7E8 ; 0104 ; A*100/255
accel_pct ; 7E8 ; 0149 ; A*100/255
volt_v ; 7E8 ; 0142 ; (A*256+B)/1000
#
# --- Блок с PID 5B (отвечает с 7EA): остаток гибридной батареи, предположительно SOC; сверить с панелью! ---
soc_5b_pct ; 7EA ; 015B ; A*100/255
#
# --- Mode 22 (проверено пользователем опытным путём), отвечает блок 7E8 ---
dis_0108_speed ; 7E8 ; 220108 ; (A*256+B)/100
dis_0103_coolant_c ; 7E8 ; 220103 ; A*3/4-48
dis_0104_rpm ; 7E8 ; 220104 ; (A*256+B)/4
#
# --- Свои Mode 22 добавлять по образцу: имя ; 7E8 ; 22XXXX ; формула ---
#
# --- Блок управления батареей BMS (запрос 7A1, ответ 7A9), Mode 22 ---
# Напряжение батареи, В (сверено: сумма 114 элементов = F228/10)
batt_v ; 7A1>7A9 ; 22F228 ; (A*256+B)/10
# Ток батареи, А (смещение 10000, шаг 0,1 А; плюс = разряд, сверено с Car Scanner в простое 0,6 А)
batt_a ; 7A1>7A9 ; 22F229 ; (A*256+B-10000)/10
# Максимальное и минимальное напряжение элемента, мВ
cell_max_mv ; 7A1>7A9 ; 22F250 ; A*256+B
cell_min_mv ; 7A1>7A9 ; 22F251 ; A*256+B
# Температура батареи max/min, °C
batt_tmax_c ; 7A1>7A9 ; 22F252 ; A-40
batt_tmin_c ; 7A1>7A9 ; 22F253 ; A-40
# Предположительно SOC в десятых долях процента (проверить по панели)
soc_f300 ; 7A1>7A9 ; 22F300 ; (A*256+B)/10
#
# --- Кандидаты (гипотезы, сверять с панелью и режимом) ---
# F22F: возможно "реальный" SOC в десятых долях %
bms_f22f ; 7A1>7A9 ; 22F22F ; (A*256+B)/10
bms_f230 ; 7A1>7A9 ; 22F230 ; (A*256+B)/10
# F258: возможно SOH, десятые доли %
bms_f258 ; 7A1>7A9 ; 22F258 ; (A*256+B)/10
bms_f264 ; 7A1>7A9 ; 22F264 ; A*256+B
# F25F: 8000, возможно порог удержания 80,00 %
bms_f25f ; 7A1>7A9 ; 22F25F ; (A*256+B)/100
